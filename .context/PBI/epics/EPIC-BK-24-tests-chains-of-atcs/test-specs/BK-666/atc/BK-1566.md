## 2. ATC Contract

```typescript
/**
 * ATC: Every Tests row carries the Test's name, its chain length in words
 * ("N ATCs" or "1 ATC"), its tags (or an em dash when none), and the UTC day
 * it was created.
 *
 * Fixed assertions:
 *  - headers Name, Chain, Tags and Created are visible
 *  - each row contains its title
 *  - each row reads "N ATCs" for N != 1 and "1 ATC" for N = 1
 *  - each row shows each of its tags, or an em dash when it has none
 *  - each row contains its UTC creation day (YYYY-MM-DD)
 *
 * @param rows - the expected rows (testId, title, chainLength, tags, createdDay)
 */
@atc('BK-1566')
async rowsShowNameChainCountTagsAndCreatedDay(rows: ProjectTestsRow[]): Promise<void>
```

`ProjectTestsRow` is declared in `ProjectTestsPage.ts`: `{ testId: string, title: string, chainLength: number, tags: string[], createdDay: string }`.

## 5. Code Template

```typescript
@atc('BK-1566')
async rowsShowNameChainCountTagsAndCreatedDay(rows: ProjectTestsRow[]): Promise<void> {
  for (const header of ['Name', 'Chain', 'Tags', 'Created']) {
    await expect(this.page.getByRole('columnheader', { name: header })).toBeVisible();
  }
  for (const row of rows) {
    const locator = this.page.locator(`[data-testid="tests-list-row-${row.testId}"]`);
    await expect(locator).toContainText(row.title);
    const chain = row.chainLength === 1 ? '1 ATC' : `${row.chainLength} ATCs`;
    await expect(locator).toContainText(chain);
    if (row.tags.length === 0) {
      await expect(locator).toContainText('\u2014');
    }
    else {
      for (const tag of row.tags) {
        await expect(locator).toContainText(tag);
      }
    }
    await expect(locator).toContainText(row.createdDay);
  }
}
```

Note: `getByRole('columnheader')` depends on the table using `<th>`. Confirm in Phase 2 (automation-plan Q8).
