## 5. Code Template

```typescript
@atc('BK-1569')
async tagFilterNarrowsListAndStatesCount(args: TagFilterCountArgs): Promise<void> {
  await this.page.locator('[data-testid="test-tag-filter"]').fill(args.tag);
  const rows = this.page.locator('[data-testid^="tests-list-row-"]');
  await expect(rows).toHaveCount(args.matchingTitles.length);
  for (const title of args.matchingTitles) {
    await expect(this.page.locator('[data-testid="tests-list"]')).toContainText(title);
  }
  const n = args.matchingTitles.length;
  const verb = n === 1 ? 'carries' : 'carry';
  await expect(this.page.locator('[data-testid="tests-list-count"]'))
    .toHaveText(`${n} of ${args.total} Tests ${verb} ${args.tag}`);
}
```
