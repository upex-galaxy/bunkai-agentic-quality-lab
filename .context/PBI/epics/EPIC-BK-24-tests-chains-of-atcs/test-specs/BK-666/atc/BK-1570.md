## 2. ATC Contract

```typescript
/**
 * ATC: Typing a tag into the Project's Tests list narrows it to this Project's
 * matching Tests. A Test with that tag in another Project is never listed, and
 * no foreign title appears.
 *
 * Fixed assertions:
 *  - the rows shown are exactly the expected titles
 *  - no foreign title appears in the list
 *
 * @param args - the tag, the expected own titles and the foreign titles that must stay out
 */
@atc('BK-1570')
async tagFilterKeepsListProjectScoped(args: TagScopeArgs): Promise<void>
```

`TagScopeArgs` is declared in `ProjectTestsPage.ts`: `{ tag: string, expectedTitles: string[], foreignTitles: string[] }`.

### Fixed (inside ATC)

### Test-level (in test file)

## 5. Code Template

```typescript
@atc('BK-1570')
async tagFilterKeepsListProjectScoped(args: TagScopeArgs): Promise<void> {
  await this.page.locator('[data-testid="test-tag-filter"]').fill(args.tag);
  const list = this.page.locator('[data-testid="tests-list"]');
  await expect(this.page.locator('[data-testid^="tests-list-row-"]')).toHaveCount(args.expectedTitles.length);
  for (const title of args.expectedTitles) {
    await expect(list).toContainText(title);
  }
  for (const title of args.foreignTitles) {
    await expect(list).not.toContainText(title);
  }
}
```
